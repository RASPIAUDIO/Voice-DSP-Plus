cmake_minimum_required(VERSION 3.16)
project(voicedsp_v2_pi48 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_library(voicedsp_pi48 STATIC
  drivers/voicedsp_v2_host.c drivers/voicedsp_v2_host_audio.c
  drivers/voicedsp_v2_control.c drivers/voicedsp_v2_power.c
  drivers/voicedsp_v2_policy.c drivers/voicedsp_v2_host_boot.c
  drivers/voicedsp_v2_host_bus.c drivers/voicedsp_v2_equalizer.c
  drivers/voicedsp_v2_gesture.c drivers/v2_ui_runtime.cpp)
target_include_directories(voicedsp_pi48 PUBLIC drivers)
if(UNIX)
  target_link_libraries(voicedsp_pi48 PUBLIC m)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(voicedsp_pi48_linux STATIC linux/transport.c)
  target_include_directories(voicedsp_pi48_linux PUBLIC linux)
  target_link_libraries(voicedsp_pi48_linux PUBLIC voicedsp_pi48)
  add_executable(voicedsp-pi48 linux/main.cpp)
  target_link_libraries(voicedsp-pi48 PRIVATE voicedsp_pi48_linux)
endif()
enable_testing()
add_executable(aec_contract tests/aec_contract.cpp)
target_link_libraries(aec_contract PRIVATE voicedsp_pi48)
if(MSVC)
  target_compile_options(aec_contract PRIVATE /UNDEBUG)
else()
  target_compile_options(aec_contract PRIVATE -UNDEBUG)
endif()
add_test(NAME accepted_aec_profiles COMMAND aec_contract)
add_executable(settings_contract tests/settings.cpp)
target_include_directories(settings_contract PRIVATE linux)
target_link_libraries(settings_contract PRIVATE voicedsp_pi48)
if(MSVC)
  target_compile_options(settings_contract PRIVATE /UNDEBUG)
else()
  target_compile_options(settings_contract PRIVATE -UNDEBUG)
endif()
add_test(NAME settings_and_gestures COMMAND settings_contract)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_test(NAME controller_help COMMAND voicedsp-pi48 --help)
endif()
